using System;
using System.Diagnostics;
using System.IO;
using System.Reflection;
using System.Runtime.InteropServices;
using System.Text;
using System.Threading;

#if LAUNCHER
[assembly: AssemblyTitle("Shelly")]
[assembly: AssemblyProduct("Shelly")]
[assembly: AssemblyDescription("Shelly - a Windows 11-style launcher for PowerShell scripts")]
[assembly: AssemblyVersion("%VERSION%")]
[assembly: AssemblyFileVersion("%VERSION%")]
#endif

static class Program
{
    const string ScriptData =
        %SCRIPT%;
    const string MsgPressKey = %MSG_PRESSKEY%;
    const string MsgTempFail = %MSG_TEMPFAIL%;
    const string MsgPsFail = %MSG_PSFAIL%;
    // Engine: "7" (PowerShell 7) or "5.1" (Windows PowerShell); "auto" = PowerShell 7 when installed.
    // The SHELLY_HOST environment variable (auto | 7 | 5.1) overrides it for Shelly.exe.
    const string Engine = %ENGINE%;
    const string MsgEngineFallback = %MSG_ENGINEFALLBACK%;

    static string tempPath;
    delegate bool ConsoleCtrlHandler(int ctrlType);
    [DllImport("kernel32.dll")]
    static extern bool SetConsoleCtrlHandler(ConsoleCtrlHandler handler, bool add);
    [DllImport("kernel32.dll")]
    static extern uint GetConsoleProcessList(uint[] list, uint count);
    static ConsoleCtrlHandler ctrlHandler;

    static string Quote(string s)
    {
        if (s.Length > 0 && s.IndexOfAny(new char[] { ' ', '\t', '"' }) < 0) return s;
        StringBuilder sb = new StringBuilder("\"");
        int bs = 0;
        foreach (char ch in s)
        {
            if (ch == '\\') { bs++; continue; }
            if (ch == '"') { sb.Append('\\', bs * 2 + 1); sb.Append('"'); }
            else { sb.Append('\\', bs); sb.Append(ch); }
            bs = 0;
        }
        sb.Append('\\', bs * 2);
        sb.Append('"');
        return sb.ToString();
    }

    static string WinPs()
    {
        string p = Path.Combine(Environment.SystemDirectory, @"WindowsPowerShell\v1.0\powershell.exe");
        return File.Exists(p) ? p : null;
    }

    static string AppPath(Microsoft.Win32.RegistryKey hive)
    {
        try
        {
            using (Microsoft.Win32.RegistryKey k = hive.OpenSubKey(@"SOFTWARE\Microsoft\Windows\CurrentVersion\App Paths\pwsh.exe"))
            {
                if (k == null) return null;
                string v = k.GetValue("") as string;
                if (v != null) v = Environment.ExpandEnvironmentVariables(v.Trim('"'));
                return (v != null && File.Exists(v)) ? v : null;
            }
        }
        catch { return null; }
    }

    // PowerShell 7: PATH, Program Files, App Paths, then the preview
    static string Pwsh()
    {
        string path = Environment.GetEnvironmentVariable("PATH") ?? "";
        foreach (string d in path.Split(';'))
        {
            try
            {
                if (d.Trim().Length == 0) continue;
                string p = Path.Combine(Environment.ExpandEnvironmentVariables(d.Trim().Trim('"')), "pwsh.exe");
                if (File.Exists(p) && p.IndexOf(@"\WindowsApps\", StringComparison.OrdinalIgnoreCase) < 0) return p;
            }
            catch { }
        }
        string pf = Environment.GetEnvironmentVariable("ProgramW6432");
        if (string.IsNullOrEmpty(pf)) pf = Environment.GetFolderPath(Environment.SpecialFolder.ProgramFiles);
        string s = Path.Combine(pf, @"PowerShell\7\pwsh.exe");
        if (File.Exists(s)) return s;
        string r = AppPath(Microsoft.Win32.Registry.LocalMachine) ?? AppPath(Microsoft.Win32.Registry.CurrentUser);
        if (r != null) return r;
        s = Path.Combine(pf, @"PowerShell\7-preview\pwsh.exe");
        if (File.Exists(s)) return s;
        // Microsoft Store install: start it through the app execution alias (the package folder itself is protected)
        string la = Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData);
        s = Path.Combine(la, @"Microsoft\WindowsApps\pwsh.exe");
        if (File.Exists(s)) return s;
        s = Path.Combine(la, @"Microsoft\WindowsApps\Microsoft.PowerShell_8wekyb3d8bbwe\pwsh.exe");
        return File.Exists(s) ? s : null;
    }

#if LAUNCHER
    static string SettingsHost()
    {
        try
        {
            string dir = Path.GetDirectoryName(Assembly.GetExecutingAssembly().Location);
            string f = Path.Combine(dir, "launcher-settings.json");
            if (!File.Exists(f) && !Directory.Exists(Path.Combine(dir, "Scripts")))
            {
                string up = Path.GetDirectoryName(dir);
                if (up != null && File.Exists(Path.Combine(up, "launcher-settings.json"))) f = Path.Combine(up, "launcher-settings.json");
            }
            if (!File.Exists(f)) return null;
            System.Text.RegularExpressions.Match m = System.Text.RegularExpressions.Regex.Match(File.ReadAllText(f),
                "\"launcherHost\"\\s*:\\s*\"(auto|7|5\\.1)\"");
            return m.Success ? m.Groups[1].Value : null;
        }
        catch { return null; }
    }
#endif

    static string PowerShellExe()
    {
        string want = Engine;
#if LAUNCHER
        string env = Environment.GetEnvironmentVariable("SHELLY_HOST");
        if (!string.IsNullOrEmpty(env))
        {
            env = env.Trim().ToLowerInvariant();
            if (env == "7" || env == "pwsh") want = "7";
            else if (env == "5.1" || env == "5" || env == "powershell") want = "5.1";
            else if (env == "auto") want = "auto";
        }
        else
        {
            // Settings > Engines > "Shelly runs on" (launcherHost in launcher-settings.json) is read at every start
            string h = SettingsHost();
            if (h != null) want = h;
        }
#endif
        string seven = Pwsh();
        string five = WinPs();
        if (want == "auto") return seven ?? five ?? "powershell.exe";
        if (want == "7")
        {
            if (seven != null) return seven;
#if !LAUNCHER
            Console.Error.WriteLine(string.Format(MsgEngineFallback, "PowerShell 7", "Windows PowerShell 5.1"));
#endif
            return five ?? "powershell.exe";
        }
        if (five != null) return five;
        if (seven != null)
        {
#if !LAUNCHER
            Console.Error.WriteLine(string.Format(MsgEngineFallback, "Windows PowerShell 5.1", "PowerShell 7"));
#endif
            return seven;
        }
        return "powershell.exe";
    }

    static string WriteScript(string dir, string fileName, byte[] data)
    {
        try { string p = Path.Combine(dir, fileName); File.WriteAllBytes(p, data); return p; }
        catch { return null; }
    }

    static void Cleanup()
    {
        string p = tempPath;
        if (p == null) return;
        for (int i = 0; i < 15; i++)
        {
            try { if (File.Exists(p)) File.Delete(p); tempPath = null; return; }
            catch { Thread.Sleep(200); }
        }
    }

    static bool OnCtrl(int ctrlType)
    {
        // Ctrl+C and Ctrl+Break are handled by PowerShell itself; on window close remove the temporary file
        if (ctrlType == 0 || ctrlType == 1) return true;
        Cleanup();
        return false;
    }

    static void Fail(string message)
    {
#if LAUNCHER
        System.Windows.Forms.MessageBox.Show(message, "Shelly", System.Windows.Forms.MessageBoxButtons.OK, System.Windows.Forms.MessageBoxIcon.Error);
#else
        Console.Error.WriteLine(message);
#endif
    }

    [STAThread]
    static int Main(string[] args)
    {
        string exePath = Assembly.GetExecutingAssembly().Location;
        string exeDir = Path.GetDirectoryName(exePath);
        string baseName = Path.GetFileNameWithoutExtension(exePath);
        byte[] data = Convert.FromBase64String(ScriptData);
        int pid = Process.GetCurrentProcess().Id;
#if LAUNCHER
        tempPath = WriteScript(Path.GetTempPath(), "Shelly-" + pid + ".ps1", data);
#else
        // The script copy is placed next to the exe, so $PSScriptRoot points to the exe folder
        tempPath = WriteScript(exeDir, "." + baseName + "-" + pid + ".ps1", data);
        if (tempPath == null) tempPath = WriteScript(Path.GetTempPath(), baseName + "-" + pid + ".ps1", data);
#endif
        if (tempPath == null) { Fail(MsgTempFail); return 1; }

        StringBuilder cmd = new StringBuilder("-NoProfile -ExecutionPolicy Bypass ");
#if LAUNCHER
        cmd.Append("-STA -WindowStyle Hidden ");
#endif
        cmd.Append("-File ").Append(Quote(tempPath));
        foreach (string a in args) cmd.Append(' ').Append(Quote(a));

        ProcessStartInfo psi = new ProcessStartInfo(PowerShellExe(), cmd.ToString());
        psi.UseShellExecute = false;
#if LAUNCHER
        psi.CreateNoWindow = true;
        psi.WorkingDirectory = exeDir;
        psi.EnvironmentVariables["LAUNCHER_HOME"] = exeDir;
        psi.EnvironmentVariables["SHELLY_APP"] = exePath;
#else
        ctrlHandler = new ConsoleCtrlHandler(OnCtrl);
        SetConsoleCtrlHandler(ctrlHandler, true);
#endif
        int code = 1;
        try
        {
            using (Process p = Process.Start(psi)) { p.WaitForExit(); code = p.ExitCode; }
        }
        catch (Exception e) { Fail(MsgPsFail + e.Message); }
        Cleanup();
#if !LAUNCHER
        // If the console window was opened just for this exe (double-click), keep it open
        try
        {
            if (GetConsoleProcessList(new uint[4], 4) == 1)
            {
                Console.WriteLine();
                Console.Write(MsgPressKey);
                Console.ReadKey(true);
            }
        }
        catch { }
#endif
        return code;
    }
}
